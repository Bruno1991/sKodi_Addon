cmake_minimum_required(VERSION 3.24)
project(mbuc_security VERSION 1.0.0 LANGUAGES CXX)
find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
add_library(mbuc_security src/aes_gcm.cpp)
target_include_directories(mbuc_security PUBLIC include)
target_compile_features(mbuc_security PUBLIC cxx_std_20)
target_link_libraries(mbuc_security PUBLIC OpenSSL::Crypto)
if(MSVC)
  target_compile_options(mbuc_security PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(mbuc_security PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(mbuc_security_tests tests/aes_gcm_test.cpp)
  target_link_libraries(mbuc_security_tests PRIVATE mbuc_security)
  add_test(NAME mbuc_security_tests COMMAND mbuc_security_tests)
endif()
